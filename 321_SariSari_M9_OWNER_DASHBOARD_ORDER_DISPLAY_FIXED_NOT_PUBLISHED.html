Published Firestore Rules V2 — reference copy

This block is synchronized with the Corrected Rules V2 published in Firebase on 17 Nov 2026. It is displayed for reference only. Editing or deploying this HTML does not publish or change Firestore Rules.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isOwner() {
      return request.auth != null
        && request.auth.uid == 'XCBkfDyQksWLLlabpCTHyh4g5Ev1';
    }

    function isAnonCustomer() {
      return request.auth != null
        && request.auth.token.firebase.sign_in_provider == 'anonymous';
    }

    function validCustomer() {
      return request.resource.data.customer is map
        && request.resource.data.customer.keys().hasOnly(['name', 'phone', 'address', 'area'])
        && request.resource.data.customer.keys().hasAll(['name', 'phone', 'address', 'area'])
        && request.resource.data.customer.name is string
        && request.resource.data.customer.name.size() > 0
        && request.resource.data.customer.name.size() <= 100
        && request.resource.data.customer.phone is string
        && request.resource.data.customer.phone.size() > 0
        && request.resource.data.customer.phone.size() <= 30
        && request.resource.data.customer.address is string
        && request.resource.data.customer.address.size() <= 300
        && (request.resource.data.customer.area == 'Store Pickup'
            || request.resource.data.customer.address.size() > 0)
        && request.resource.data.customer.area in [
          'San Carlos City Proper',
          'Vallehermoso',
          'Canlaon City',
          'Don Salvador Benedicto (DSB)',
          'Calatrava',
          'Toboso',
          'Store Pickup'
        ];
    }

    function validItem(i) {
      return i is map
        && i.keys().hasOnly(['productId', 'name', 'unit', 'quantity', 'unitPrice', 'lineTotal'])
        && i.keys().hasAll(['productId', 'name', 'unit', 'quantity', 'unitPrice', 'lineTotal'])
        && i.productId is string
        && i.productId.size() > 0
        && i.productId.size() <= 128
        && i.name is string
        && i.name.size() > 0
        && i.name.size() <= 150
        && i.unit is string
        && i.unit.size() > 0
        && i.unit.size() <= 30
        && i.quantity is int
        && i.quantity >= 1
        && i.quantity <= 99
        && i.unitPrice is number
        && i.unitPrice >= 0
        && i.lineTotal is number
        && i.lineTotal >= (i.unitPrice * i.quantity) - 0.000001
        && i.lineTotal <= (i.unitPrice * i.quantity) + 0.000001
        && exists(/databases/$(database)/documents/publicCatalog/$(i.productId))
        && get(/databases/$(database)/documents/publicCatalog/$(i.productId)).data.active == true
        && get(/databases/$(database)/documents/publicCatalog/$(i.productId)).data.price == i.unitPrice
        && get(/databases/$(database)/documents/publicCatalog/$(i.productId)).data.name == i.name
        && get(/databases/$(database)/documents/publicCatalog/$(i.productId)).data.unit == i.unit;
    }

    function validItems() {
      return request.resource.data.items is list
        && request.resource.data.items.size() >= 1
        && request.resource.data.items.size() <= 5
        && validItem(request.resource.data.items[0])
        && (request.resource.data.items.size() < 2 || validItem(request.resource.data.items[1]))
        && (request.resource.data.items.size() < 3 || validItem(request.resource.data.items[2]))
        && (request.resource.data.items.size() < 4 || validItem(request.resource.data.items[3]))
        && (request.resource.data.items.size() < 5 || validItem(request.resource.data.items[4]));
    }

    function validSubtotal() {
      let items = request.resource.data.items;
      let expected = items[0].lineTotal
        + (items.size() > 1 ? items[1].lineTotal : 0)
        + (items.size() > 2 ? items[2].lineTotal : 0)
        + (items.size() > 3 ? items[3].lineTotal : 0)
        + (items.size() > 4 ? items[4].lineTotal : 0);
      return request.resource.data.subtotal is number
        && request.resource.data.subtotal >= 0
        && request.resource.data.subtotal >= expected - 0.000001
        && request.resource.data.subtotal <= expected + 0.000001;
    }

    function validOrderCreate() {
      return isAnonCustomer()
        && request.resource.data.keys().hasOnly([
          'customerUid', 'customer', 'items', 'subtotal', 'paymentMethod',
          'paymentStatus', 'notes', 'deliveryFee', 'status', 'riderId',
          'riderName', 'salesAttributionStatus', 'createdAt', 'source'
        ])
        && request.resource.data.keys().hasAll([
          'customerUid', 'customer', 'items', 'subtotal', 'paymentMethod',
          'paymentStatus', 'notes', 'deliveryFee', 'status', 'riderId',
          'riderName', 'salesAttributionStatus', 'createdAt', 'source'
        ])
        && request.resource.data.customerUid == request.auth.uid
        && request.resource.data.source == 'online'
        && request.resource.data.status == 'New'
        && request.resource.data.paymentStatus == 'Pending'
        && request.resource.data.paymentMethod in ['COD', 'GCash']
        && validCustomer()
        && validItems()
        && validSubtotal()
        && request.resource.data.deliveryFee is number
        && request.resource.data.deliveryFee == 0
        && request.resource.data.notes is string
        && request.resource.data.notes.size() <= 500
        && request.resource.data.riderId == ''
        && request.resource.data.riderName == ''
        && request.resource.data.salesAttributionStatus == 'Pending owner review'
        && request.resource.data.createdAt == request.time;
    }

    match /publicCatalog/{productId} {
      allow read: if isAnonCustomer() && resource.data.active == true;
      allow create, update, delete: if isOwner();
    }

    match /customerOrders/{orderId} {
      allow create: if validOrderCreate();
      allow read: if isOwner()
        || (request.auth != null && resource.data.customerUid == request.auth.uid);
      // REVIEW NOTE: owner-only update/delete remains intentionally broad in this draft.
      // The owner UI currently sends several different patch shapes. Do not treat UI
      // restrictions as a substitute for Rules-level field/state-transition controls.
      allow update, delete: if isOwner();
    }

    match /{document=**} {
      allow read, write: if isOwner();
    }
  }
}
